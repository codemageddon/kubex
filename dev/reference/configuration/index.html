    """Connection and credential settings for a client.

    Args:
        tls_server_name: Name to verify the server certificate against (and
            send as SNI) instead of the hostname in ``url``; kubeconfig's
            ``tls-server-name``.
        username: Basic-auth username; used only when no token source is set.
        password: Basic-auth password.
        impersonate_user: Username to act as (``Impersonate-User``).
        impersonate_uid: UID to act as (``Impersonate-Uid``).
        impersonate_groups: Groups to act as (one ``Impersonate-Group`` each).
        impersonate_extra: Extra user info to act with
            (``Impersonate-Extra-<key>``, one header per value).
    """
    if try_refresh_token and token_file is None:
        raise ValueError("Token file must be provided to refresh token")
    self.refreshable_token = refreshable_token
    self.base_url = url
    if insecure_skip_tls_verify is False:
        if server_ca_file is None:
            raise ValueError("Server CA file must be provided")
    self.insecure_skip_tls_verify = insecure_skip_tls_verify
    if isinstance(server_ca_file, str):
        server_ca_file = Path(server_ca_file)
    self.server_ca_file = server_ca_file
    if isinstance(client_cert_file, str):
        client_cert_file = Path(client_cert_file)
    self.client_cert_file = client_cert_file
    if isinstance(client_key_file, str):
        client_key_file = Path(client_key_file)
    self.client_key_file = client_key_file
    self.namespace = namespace

    if isinstance(token_file, str):
        token_file = Path(token_file)

    self.token_file = token_file
    self.try_refresh_token = try_refresh_token
    self._last_token_read: float | None = None
    self._current_token: str | None = None
    self._token = token
    self.tls_server_name = tls_server_name or None
    if (username is None) != (password is None):
        raise ValueError("username and password must be provided together")
    self.username = username
    self._password = password
    if impersonate_user is None and (impersonate_groups or impersonate_extra):
        # The API server rejects group/extra impersonation without a user.
        raise ValueError(
            "impersonate_groups/impersonate_extra require impersonate_user"
        )
    self.impersonate_user = impersonate_user
    self.impersonate_uid = impersonate_uid
    self.impersonate_groups: tuple[str, ...] = tuple(impersonate_groups or ())
    self.impersonate_extra: dict[str, tuple[str, ...]] = {
        key: tuple(values) for key, values in (impersonate_extra or {}).items()

    async def get_authorization_header(self) -> str | None:
    """Return the `Authorization` header value to send, if any.

    Prefers `refreshable_token` (exec/OIDC credential plugins) over the
    static `token`/`token_file` path, since a kubeconfig context only
    ever sets one credential source.
    """
    if self.refreshable_token is not None:
        return await self.refreshable_token.to_header()
    token = self.token
    if token is not None:
        return f"Bearer {token}"
    if self.username is not None and self._password is not None:
        credentials = f"{self.username}:{self._password}".encode()
        return f"Basic {base64.b64encode(credentials).decode('ascii')}"
    return None

    def describe_validation_error(exc: ValidationError, data: typing.Any = None) -> str:
    """Summarise ``exc`` by error locations and types only.

    pydantic's own message embeds the offending input, which for kubeconfig
    and exec-plugin data can be a credential. Pass the validated ``data`` to
    name list entries (e.g. ``users['dev']``) instead of numbering them.
    """
    details = [
        f"{_format_location(error['loc'], data)} ({error['type']})"
        for error in exc.errors(include_url=False, include_input=False)
    ]
    return f"{exc.error_count()} validation error(s): {', '.join(details)}"

    async def configure_from_kubeconfig(
    config: KubeConfig | None = None,
    use_context: str | None = None,
    *,
    config_file: str | os.PathLike[str] | None = None,
) -> ClientConfiguration:
    """Create a ClientConfiguration from a kubeconfig.

    Only the selected context's cluster and user are checked (server URL,
    referenced files), so a stale entry for another context does not matter.

    Args:
        config: An already-parsed kubeconfig. Relative file paths inside it are
            used as-is (relative to the current directory); use `config_file`
            to load a file from disk.
        use_context: Context to use instead of `current-context`.
        config_file: Kubeconfig file to load instead of `$KUBECONFIG` /
            `~/.kube/config`. Relative paths inside it resolve against its
            directory, as with kubectl.

    Raises:
        FileNotFoundError: No kubeconfig file exists.
        ConfigurationError: The kubeconfig, or the selected cluster/user, is
            invalid (e.g. a referenced certificate file is missing).
        ConfigurationValueError: The context, or the cluster/user it names, is
            not defined. Also a `ConfigurationError` and a `ValueError`.
        ValueError: Both `config` and `config_file` are given.
    """
    if config is not None and config_file is not None:
        raise ValueError("Pass either config or config_file, not both")
    if config is None:
        config = _load_kube_config(
            Path(config_file) if config_file is not None else None
        )
    current_context = use_context or config.current_context
    if not current_context:
        raise ConfigurationValueError("No current context in kubeconfig")
    context = next(
        (c.context for c in config.contexts if c.name == current_context),
        None,
    )
    if context is None:
        raise ConfigurationValueError(
            f"Context {current_context} not found in kubeconfig"
        )
    if not context.cluster:
        raise ConfigurationError(
            f"kubeconfig context {current_context!r} does not name a cluster"
        )
    cluster = next(
        (c.cluster for c in config.clusters if c.name == context.cluster),
        None,
    )
    if cluster is None:
        raise ConfigurationValueError(
            f"Cluster {context.cluster} not found in kubeconfig"
        )
    user = AuthInfo()
    if context.user:
        found_user = next(
            (u.auth_info for u in config.users if u.name == context.user),
            None,
        )
        if found_user is None:
            raise ConfigurationValueError(
                f"User {context.user} not found in kubeconfig"
            )
        user = found_user
    cluster_owner = f"cluster {context.cluster!r}"
    user_owner = f"user {context.user!r}"
    server = _server_url(context.cluster, cluster.server)
    ca_file = _data_or_file(
        cluster.certificate_authority_data,
        cluster.certificate_authority,
        cluster_owner,
        "certificate-authority",
    )
    client_cert_file = _data_or_file(
        user.client_certificate_data,
        user.client_certificate,
        user_owner,
        "client-certificate",
    )
    client_key_file = _data_or_file(
        user.client_key_data, user.client_key, user_owner, "client-key"
    )
    if user.token_file is not None and not user.token:
        _require_file(user.token_file, user_owner, "tokenFile")
    refreshable_token: SupportsAuthorizationHeader | None = None
    if user.exec is not None:
        cluster_info = (
            _exec_cluster_info(server, cluster, ca_file)
            if user.exec.provide_cluster_info
            else None
        )
        refreshable_token = ExecRefreshableToken(
            ExecAuthProvider(user.exec, cluster_info=cluster_info)
        )
    elif user.auth_provider is not None:
        if user.auth_provider.name != "oidc":
            raise ConfigurationError(
                f"configure_from_kubeconfig() does not support the "
                f"{user.auth_provider.name!r} auth-provider; only 'oidc' with a "
                "valid OIDC config is supported."
            )
        refreshable_token = OidcRefreshableToken(
            OIDCAuthProvider(_oidc_config(context.user or "", user.auth_provider))
        )
    # Basic auth needs a username; client-go sends an empty password if unset.
    username = user.username or None
    password = None
    if username is not None:
        password = user.password.get_secret_value() if user.password else ""
    if cluster.proxy_url:
        warnings.warn(
            f"kubeconfig cluster {context.cluster!r} sets proxy-url, which "
            "configure_from_kubeconfig() does not apply (it belongs on "
            "ClientOptions, which this function has no access to). Pass the "
            "equivalent proxy via ClientOptions(proxy=...) when constructing "
            "the client.",
            UserWarning,
            stacklevel=2,
        )
    return ClientConfiguration(
        url=server,
        server_ca_file=ca_file,
        # Cluster.insecure_skip_tls_verify defaults to False (not None), and
        # ClientConfiguration raises when insecure_skip_tls_verify is False
        # (rather than None) with no server_ca_file -- so only forward True,
        # leaving the common "no CA, no insecure flag" kubeconfig (system
        # trust store) as ClientConfiguration's own default of None.
        insecure_skip_tls_verify=True if cluster.insecure_skip_tls_verify else None,
        client_cert_file=client_cert_file,
        client_key_file=client_key_file,
        token=user.token or None,
        token_file=user.token_file,
        # kubectl/client-go use "default" when the context sets no namespace.
        namespace=context.namespace or DEFAULT_NAMESPACE,
        try_refresh_token=user.token_file is not None,
        refreshable_token=refreshable_token,
        tls_server_name=cluster.tls_server_name,
        username=username,
        password=password,
        impersonate_user=user.as_,
        impersonate_uid=user.as_uid,
        impersonate_groups=user.as_groups,
        impersonate_extra=user.as_user_extra,
    )

    Return a usable id_token, redeeming the refresh token only when needed.

    Like client-go, the configured id-token is returned as long as its exp claim is in the future; otherwise the refresh token is redeemed for a new id_token (rotating the in-memory refresh token if the IdP issues a new one). Raises ConfigurationError when there is neither a valid id-token nor a refresh token.

    Returns the id_token and its own exp claim (Unix epoch seconds), if it is a decodable JWT carrying one. This is deliberately not the token response's expires_in field, which describes the access token's lifetime -- a separate token with its own, independently configured expiry -- not the id_token that kubex actually sends to the API server.

    A rotated refresh token is kept only on self.config for the life of this process; it is not written back to the source kubeconfig file. Against an IdP configured for refresh-token rotation (e.g. Auth0, Okta, Keycloak with rotation enabled), the on-disk kubeconfig's refresh token is single-use and is invalidated by the first successful refresh, so a fresh process (or a concurrent kubectl using the same kubeconfig) must re-authenticate interactively once this token has been used here.

    async def refresh_token(self) -> tuple[str, float | None]:
    """Return a usable id_token, redeeming the refresh token only when needed.

    Like client-go, the configured `id-token` is returned as long as its `exp`
    claim is in the future; otherwise the refresh token is redeemed for a new
    id_token (rotating the in-memory refresh token if the IdP issues a new one).
    Raises `ConfigurationError` when there is neither a valid id-token nor a
    refresh token.

    Returns the id_token and its own `exp` claim (Unix epoch seconds), if it is a
    decodable JWT carrying one. This is deliberately not the token response's
    `expires_in` field, which describes the access token's lifetime -- a
    separate token with its own, independently configured expiry -- not the
    id_token that kubex actually sends to the API server.

    A rotated refresh token is kept only on `self.config` for the life of this
    process; it is not written back to the source kubeconfig file. Against an
    IdP configured for refresh-token rotation (e.g. Auth0, Okta, Keycloak with
    rotation enabled), the on-disk kubeconfig's refresh token is single-use and
    is invalidated by the first successful refresh, so a fresh process (or a
    concurrent `kubectl` using the same kubeconfig) must re-authenticate
    interactively once this token has been used here.
    """
    configured = self._configured_id_token()
    if configured is not None:
        return configured
    if self.config.refresh_token is None:
        raise ConfigurationError(
            "OIDC: no valid id-token (missing, expired or without an exp claim) "
            "and no refresh-token to obtain a new one; log in again to refresh "
            "the kubeconfig credentials"
        )
    content = await self._id_token()
    data = json.loads(content)
    id_token = data.get("id_token")
    if not isinstance(id_token, str):
        raise ConfigurationValueError("OIDC: id_token not found in token response")
    self.config.id_token = SecretStr(id_token)
    rotated_refresh_token = data.get("refresh_token")
    if isinstance(rotated_refresh_token, str) and rotated_refresh_token:
        self.config.refresh_token = SecretStr(rotated_refresh_token)
    return id_token, _jwt_exp_claim(id_token)
